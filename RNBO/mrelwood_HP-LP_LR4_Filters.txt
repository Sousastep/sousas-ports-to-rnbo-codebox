function fn_lp_LR4ini(arg_lphp, arg_freq, arg_Q) {
s_lp_lphp = arg_lphp;
s_lp_w0 = eelDivide(((2 * PI) * arg_freq),samplerate());
let tmp28 = s_lp_w0;
s_lp_cosw = cos(tmp28);
let tmp29 = s_lp_w0;
s_lp_sinw = sin(tmp29);
s_lp_alpha = eelDivide(s_lp_sinw,(2 * arg_Q));
let tmp30 = 0;
if (arg_lphp == 0) {
s_lp_b0_lp = eelDivide((1 - s_lp_cosw),2);
s_lp_b1_lp = (1 - s_lp_cosw);
s_lp_b2_lp = eelDivide((1 - s_lp_cosw),2);
s_lp_a0_lp = (1 + s_lp_alpha);
s_lp_a1_lp = ((0 - 2) * s_lp_cosw);
s_lp_a2_lp = (1 - s_lp_alpha);
s_lp_b0_lp /= s_lp_a0_lp;
s_lp_b1_lp /= s_lp_a0_lp;
s_lp_b2_lp /= s_lp_a0_lp;
s_lp_a1_lp /= s_lp_a0_lp;
s_lp_a2_lp /= s_lp_a0_lp;
tmp30 = s_lp_a2_lp;
} else {
s_lp_b0_hp = eelDivide((1 + s_lp_cosw),2);
s_lp_b1_hp = (0 - (1 + s_lp_cosw));
s_lp_b2_hp = eelDivide((1 + s_lp_cosw),2);
s_lp_a0_hp = (1 + s_lp_alpha);
s_lp_a1_hp = ((0 - 2) * s_lp_cosw);
s_lp_a2_hp = (1 - s_lp_alpha);
s_lp_b0_hp /= s_lp_a0_hp;
s_lp_b1_hp /= s_lp_a0_hp;
s_lp_b2_hp /= s_lp_a0_hp;
s_lp_a1_hp /= s_lp_a0_hp;
s_lp_a2_hp /= s_lp_a0_hp;
tmp30 = s_lp_a2_hp;
}
return tmp30;
}

}
