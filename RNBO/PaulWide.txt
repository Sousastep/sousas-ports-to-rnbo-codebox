// VST A: Quant
@param({ min: 0, max: 1 }) p_quant = 1.0;
// VST B: DeRez
@param({ min: 0, max: 1 }) p_derez = 0.0;

@state fpdL = 305419896.0;
@state fpdR = 2271560481.0;
@state previousDitherL = 0;
@state previousDitherR = 0;
@state processing: Int = 0;
@state highres = 0;
@state derez = 0;
@state outScale = 0;
@state scaleFactor = 0;
@state frame: Int = 0;
// Xorshift32 with explicit unsigned wrapping (fixed, reproducible seeds).
function unsigned32(x) {
    return x - floor(x / 4294967296.0) * 4294967296.0;
}
function advanceRandom(x) {
    let a: Int = x;
    let left: Int = unsigned32(x * 8192.0);
    let b: Int = unsigned32(a ^ left);
    let shifted: Int = floor(b / 131072.0);
    let c: Int = unsigned32(b ^ shifted);
    left = unsigned32(c * 32.0);
    return unsigned32(c ^ left);
}
function toIndex(x) {
    let result: Int = trunc(x);
    return result;
}

@state floatCast = new FixedFloat32Array(1);
function asFloat32(x) { floatCast[0] = x; return floatCast[0]; }

function init() {

	previousDitherL = 0.0;
	previousDitherR = 0.0;

}
function dspsetup() { frame = 0; }

if (frame == 0) {
processing = toIndex( p_quant * 1.999 );
	highres = 0;
	if (processing == 1) highres = 1;
	
	if (highres) scaleFactor = 8388608.0;
	else scaleFactor = 32768.0;
	derez = p_derez;
	if (derez > 0.0) scaleFactor = asFloat32(scaleFactor * pow(1.0-derez,6));
	if (scaleFactor < 0.0001) scaleFactor = asFloat32(0.0001);
	outScale = scaleFactor;
	if (outScale < 8.0) outScale = 8.0;

}
let sampleFrames = vectorsize() - frame - 1;
let inFramesToProcess = vectorsize();

		let inputSampleL = in1;
		let inputSampleR = in2;
		if (abs(inputSampleL)<1.18e-23) inputSampleL = fpdL * 1.18e-17;
		fpdL = advanceRandom(fpdL);
		if (abs(inputSampleR)<1.18e-23) inputSampleR = fpdR * 1.18e-17;
		fpdR = advanceRandom(fpdR);
		
		inputSampleL *= scaleFactor;
		inputSampleR *= scaleFactor;

		let currentDither = ((fpdL)/4294967295.0);
		let ditherL = currentDither;
		ditherL -= previousDitherL;
		previousDitherL = currentDither;

		currentDither = ((fpdR)/4294967295.0);
		let ditherR = currentDither;
		ditherR -= previousDitherR;
		previousDitherR = currentDither;

		if (abs(ditherL-ditherR) < 0.5) {
			fpdL = advanceRandom(fpdL);
			currentDither = ((fpdL)/4294967295.0);
			ditherL = currentDither;
			ditherL -= previousDitherL;
			previousDitherL = currentDither;
		}
		
		if (abs(ditherL-ditherR) < 0.5) {
			fpdR = advanceRandom(fpdR);
			currentDither = ((fpdR)/4294967295.0);
			ditherR = currentDither;
			ditherR -= previousDitherR;
			previousDitherR = currentDither;
		}
		
		if (abs(ditherL-ditherR) < 0.5) {
			fpdL = advanceRandom(fpdL);
			currentDither = ((fpdL)/4294967295.0);
			ditherL = currentDither;
			ditherL -= previousDitherL;
			previousDitherL = currentDither;
		}
		
		inputSampleL = floor(inputSampleL+ditherL);
		inputSampleR = floor(inputSampleR+ditherR);
		
		inputSampleL /= outScale;
		inputSampleR /= outScale;
		
		fpdL = advanceRandom(fpdL);
		fpdR = advanceRandom(fpdR);

out1 = inputSampleL;
out2 = inputSampleR;
frame++;
if (frame >= vectorsize()) frame = 0;
