// VST A: Boost
@param({ min: 0, max: 1 }) p_boost = 0.0;
// VST B: Match
@param({ min: 0, max: 1 }) p_match = 0.0;
// VST C: Noise
@param({ min: 0, max: 1 }) p_noise = 0.7;
// VST D: Ceiling
@param({ min: 0, max: 1 }) p_ceiling = 0.75;
// VST E: Mode
@param({ min: 0, max: 1 }) p_mode = 0.0;

@state lastSampleL = 0;
@state lastDryL = 0;
@state intermediateL = new FixedFloat64Array(18);
@state slewL = new FixedFloat64Array(34);
@state wasPosClipL = 0;
@state wasNegClipL = 0;
@state lastSampleR = 0;
@state lastDryR = 0;
@state intermediateR = new FixedFloat64Array(18);
@state slewR = new FixedFloat64Array(34);
@state wasPosClipR = 0;
@state wasNegClipR = 0;
@state fpdL = 305419896.0;
@state fpdR = 2271560481.0;
@state overallscale = 0;
@state spacing: Int = 0;
@state inputGain = 0;
@state match = 0;
@state noiseLevel = 0;
@state ceiling = 0;
@state mode: Int = 0;
@state frame: Int = 0;
// Xorshift32 with explicit unsigned wrapping (fixed, reproducible seeds).
function unsigned32(x) {
    return x - floor(x / 4294967296.0) * 4294967296.0;
}
function advanceRandom(x) {
    let a: Int = x;
    let left: Int = unsigned32(x * 8192.0);
    let b: Int = unsigned32(a ^ left);
    let shifted: Int = floor(b / 131072.0);
    let c: Int = unsigned32(b ^ shifted);
    left = unsigned32(c * 32.0);
    return unsigned32(c ^ left);
}
function toIndex(x) {
    let result: Int = trunc(x);
    return result;
}

function init() {

	lastSampleL = 0.0;
	wasPosClipL = 0;
	wasNegClipL = 0;
	lastSampleR = 0.0;
	wasPosClipR = 0;
	wasNegClipR = 0;
	for (let x: Int = 0; x < 17; x++) {intermediateL[x] = 0.0; intermediateR[x] = 0.0;}
	for (let x: Int = 0; x < 33; x++) {slewL[x] = 0.0; slewR[x] = 0.0;}

}
function dspsetup() { frame = 0; }

if (frame == 0) {
overallscale = 1.0;
	overallscale /= 44100.0;
	overallscale *= samplerate();

	spacing = floor(overallscale); 
	if (spacing < 1) spacing = 1; if (spacing > 16) spacing = 16;
	
	inputGain = pow(10.0,(p_boost*18.0)/20.0);
	match = pow(10.0,(p_match*18.0)/20.0);
	noiseLevel = 1.0-pow(1.0-p_noise,2.0);
	ceiling = 1.0-pow(1.0-p_ceiling,2.0);
	mode = toIndex(p_mode*2.999)+1;

}
let sampleFrames = vectorsize() - frame - 1;
let inFramesToProcess = vectorsize();

		let inputSampleL = in1;
		let inputSampleR = in2;
		if (abs(inputSampleL)<1.18e-23) inputSampleL = fpdL * 1.18e-17;
		if (abs(inputSampleR)<1.18e-23) inputSampleR = fpdR * 1.18e-17;
		
		switch (mode)
		{
			case 1: inputSampleL *= inputGain; inputSampleR *= inputGain; break; 
			case 2: inputSampleL *= match; inputSampleR *= match; break; 
			case 3: inputSampleL *= inputGain; inputSampleR *= inputGain; break; 
		}

		let overshootL = lastDryL;
		lastDryL = inputSampleL;
		let overshootR = lastDryR;
		lastDryR = inputSampleR;

		let noise = 1.0-(((fpdL)/4294967295.0)*(1.0-noiseLevel));
		if (wasPosClipL == 1) { 
			if (inputSampleL<lastSampleL) lastSampleL=(noiseLevel*noise)+(inputSampleL*(1.0-noise));
			else lastSampleL = ceiling;
		} wasPosClipL = 0;
		if (inputSampleL>noiseLevel) {wasPosClipL=1;inputSampleL=(noiseLevel*noise)+(lastSampleL*(1.0-noise));}
		if (wasNegClipL == 1) { 
			if (inputSampleL > lastSampleL) lastSampleL=(-noiseLevel*noise)+(inputSampleL*(1.0-noise));
			else lastSampleL = -ceiling;
		} wasNegClipL = 0;
		if (inputSampleL<-noiseLevel) {wasNegClipL=1;inputSampleL=(-noiseLevel*noise)+(lastSampleL*(1.0-noise));}
		slewL[spacing*2] = abs(lastSampleL-inputSampleL);
		for (let x: Int = spacing*2; x > 0; x--) slewL[x-1] = slewL[x];
		intermediateL[spacing] = inputSampleL; inputSampleL = lastSampleL;
		
		for (let x: Int = spacing; x > 0; x--) {intermediateL[x-1] = intermediateL[x];} lastSampleL = intermediateL[0];
		if (wasPosClipL || wasNegClipL) {
			for (let x: Int = spacing; x > 0; x--) lastSampleL += intermediateL[x];
			lastSampleL /= spacing;
		} let finalSlew = 0.0;
		for (let x: Int = spacing*2; x >= 0; x--) if (finalSlew < slewL[x]) finalSlew = slewL[x];
		let postclip = ceiling / (1.0+(finalSlew*1.3986013));
		if (inputSampleL > postclip) inputSampleL = postclip; if (inputSampleL < -postclip) inputSampleL = -postclip;
		
		noise = 1.0-(((fpdR)/4294967295.0)*(1.0-noiseLevel));
		if (wasPosClipR == 1) { 
			if (inputSampleR<lastSampleR) lastSampleR=(noiseLevel*noise)+(inputSampleR*(1.0-noise));
			else lastSampleR = ceiling;
		} wasPosClipR = 0;
		if (inputSampleR>noiseLevel) {wasPosClipR=1;inputSampleR=(noiseLevel*noise)+(lastSampleR*(1.0-noise));}
		if (wasNegClipR == 1) { 
			if (inputSampleR > lastSampleR) lastSampleR=(-noiseLevel*noise)+(inputSampleR*(1.0-noise));
			else lastSampleR = -ceiling;
		} wasNegClipR = 0;
		if (inputSampleR<-noiseLevel) {wasNegClipR=1;inputSampleR=(-noiseLevel*noise)+(lastSampleR*(1.0-noise));}
		slewR[spacing*2] = abs(lastSampleR-inputSampleR);
		for (let x: Int = spacing*2; x > 0; x--) slewR[x-1] = slewR[x];
		intermediateR[spacing] = inputSampleR; inputSampleR = lastSampleR;
		
		for (let x: Int = spacing; x > 0; x--) {intermediateR[x-1] = intermediateR[x];} lastSampleR = intermediateR[0];
		if (wasPosClipR || wasNegClipR) {
			for (let x: Int = spacing; x > 0; x--) lastSampleR += intermediateR[x];
			lastSampleR /= spacing;
		} finalSlew = 0.0;
		for (let x: Int = spacing*2; x >= 0; x--) if (finalSlew < slewR[x]) finalSlew = slewR[x];
		postclip = ceiling / (1.0+(finalSlew*1.3986013));
		if (inputSampleR > postclip) inputSampleR = postclip; if (inputSampleR < -postclip) inputSampleR = -postclip;

		switch (mode)
		{
			case 1: break; 
			case 2: inputSampleL /= match; inputSampleR /= match; break; 
			case 3: inputSampleL = (inputSampleL-overshootL)/inputGain; inputSampleR = (inputSampleR-overshootR)/inputGain; break; 
		}

		fpdL = advanceRandom(fpdL);

		fpdR = advanceRandom(fpdR);
		if (unsigned32(fpdL-fpdR) < 1073741824 || unsigned32(fpdR-fpdL) < 1073741824) {
			fpdR = advanceRandom(fpdR);}

out1 = inputSampleL;
out2 = inputSampleR;
frame++;
if (frame >= vectorsize()) frame = 0;
